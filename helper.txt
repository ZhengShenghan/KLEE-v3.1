# dependencies
sudo apt-get update
sudo apt-get install -y \
  clang-13 llvm-13 llvm-13-dev llvm-13-tools \
  build-essential cmake git python3 python3-pip \
  z3 libz3-dev \
  libcap-dev libelf-dev \
  libgoogle-perftools-dev


# klee-uclibc
./configure --make-llvm-lib --with-llvm-config=/usr/bin/llvm-config-13

make

## sqlite3
sudo apt-get update
sudo apt-get install -y libsqlite3-dev sqlite3

# stp(optional)
cd ~/klee_deps/stp/build
rm -rf ./*

cmake .. \
  -DCMAKE_BUILD_TYPE=Release \
  -DBUILD_SHARED_LIBS=ON \
  -DENABLE_TESTING=OFF \
  -DENABLE_PYTHON_INTERFACE=OFF \
  -DENABLE_ABC=OFF \
  -DMINISAT_INCLUDE_DIR=$HOME/klee_deps/minisat \
  -DMINISAT_LIBRARY=$HOME/klee_deps/minisat-build/libminisat.so \
  -DCMAKE_INSTALL_PREFIX=$HOME/klee_deps/stp-install

make -j"$(nproc)"
make install

# build klee
cd ~/Documents/klee/build
rm -rf ./*

cmake .. \
  -DLLVM_CONFIG_BINARY=/usr/bin/llvm-config-13 \
  -DENABLE_SOLVER_STP=OFF \
  -DENABLE_SOLVER_Z3=ON \
  -DSOLVER_DEFAULT=Z3 \
  -DENABLE_POSIX_RUNTIME=ON \
  -DENABLE_TCMALLOC=ON \
  -DKLEE_UCLIBC_PATH=../../klee-uclibc
